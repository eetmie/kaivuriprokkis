cmake_minimum_required(VERSION 3.16)
project(kaivuri_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_CAMERA "Build the RealSense capture service" ON)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
add_library(kaivuri_core src/core.cpp)
target_include_directories(kaivuri_core PUBLIC include)
target_link_libraries(kaivuri_core PUBLIC yaml-cpp Threads::Threads)
target_compile_options(kaivuri_core PUBLIC -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
    target_compile_options(kaivuri_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kaivuri_core PUBLIC -fsanitize=address,undefined)
endif()
add_executable(kaivuri-control src/control_main.cpp)
target_link_libraries(kaivuri-control PRIVATE kaivuri_core)
add_library(kaivuri_frames SHARED src/frame_store.cpp)
target_include_directories(kaivuri_frames PUBLIC include)
target_link_libraries(kaivuri_frames PRIVATE Threads::Threads)
target_compile_options(kaivuri_frames PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
    target_compile_options(kaivuri_frames PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kaivuri_frames PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_CAMERA)
    find_package(realsense2 2.58.4 EXACT CONFIG REQUIRED)
    add_executable(kaivuri-camera src/camera_main.cpp)
    target_link_libraries(kaivuri-camera PRIVATE kaivuri_frames realsense2::realsense2)
    target_compile_options(kaivuri-camera PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_executable(core-tests tests/core_tests.cpp)
target_link_libraries(core-tests PRIVATE kaivuri_core)
target_compile_definitions(core-tests PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME core COMMAND core-tests ${CMAKE_CURRENT_SOURCE_DIR}/../configuration_files/profiles/jetson)
add_executable(frame-tests tests/frame_tests.cpp)
target_link_libraries(frame-tests PRIVATE kaivuri_frames Threads::Threads)
add_test(NAME frame-store COMMAND frame-tests)
add_executable(reference-probe tests/reference_probe.cpp)
target_link_libraries(reference-probe PRIVATE kaivuri_core)
target_compile_definitions(reference-probe PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME service-integration COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:kaivuri-control> ${CMAKE_CURRENT_SOURCE_DIR})
