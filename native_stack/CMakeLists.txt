cmake_minimum_required(VERSION 3.16)
project(kaivuri_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_CAMERA "Build the RealSense capture service" ON)
option(BUILD_INFERENCE "Build the OpenPI pi0.5 TensorRT bench runtime" OFF)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
find_package(yaml-cpp REQUIRED)
find_package(Threads REQUIRED)
# Lossless depth compression in the recorder spool and camera depth log.
find_package(PkgConfig REQUIRED)
pkg_check_modules(ZSTD REQUIRED IMPORTED_TARGET libzstd)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
add_library(kaivuri_core src/core.cpp src/util.cpp)
target_include_directories(kaivuri_core PUBLIC include)
target_link_libraries(kaivuri_core PUBLIC yaml-cpp Threads::Threads OpenSSL::Crypto)
target_compile_options(kaivuri_core PUBLIC -Wall -Wextra -Wpedantic)
add_library(kaivuri_io src/pwm.cpp src/gamepad.cpp)
target_link_libraries(kaivuri_io PUBLIC kaivuri_core)
if(ENABLE_SANITIZERS)
    target_compile_options(kaivuri_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kaivuri_core PUBLIC -fsanitize=address,undefined)
endif()
# CUDA-free pi0.5 prompt/normalization/producer logic; exact float64 order, no FMA contraction.
add_library(kaivuri_pi05 src/pi05_policy.cpp)
target_link_libraries(kaivuri_pi05 PUBLIC kaivuri_core)
target_compile_options(kaivuri_pi05 PRIVATE -ffp-contract=off)
add_executable(kaivuri-control src/control_main.cpp)
target_link_libraries(kaivuri-control PRIVATE kaivuri_io)
add_library(kaivuri_frames SHARED src/frame_store.cpp)
target_include_directories(kaivuri_frames PUBLIC include)
target_link_libraries(kaivuri_frames PRIVATE Threads::Threads)
target_compile_options(kaivuri_frames PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
    target_compile_options(kaivuri_frames PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(kaivuri_frames PUBLIC -fsanitize=address,undefined)
endif()
add_executable(kaivuri-record src/record_main.cpp)
target_link_libraries(kaivuri-record PRIVATE kaivuri_io kaivuri_frames PkgConfig::ZSTD)
add_library(kaivuri_mlp src/mlp_excitation.cpp src/mlp_record.cpp)
target_link_libraries(kaivuri_mlp PUBLIC kaivuri_io)
set(NATIVE_GIT_REVISION "" CACHE STRING "Source revision for an exported source tree")
if(NOT NATIVE_GIT_REVISION)
    execute_process(COMMAND git -C ${CMAKE_CURRENT_SOURCE_DIR} rev-parse HEAD OUTPUT_VARIABLE NATIVE_GIT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
add_executable(kaivuri-mlp-collect src/mlp_main.cpp src/mlp_options.cpp)
target_link_libraries(kaivuri-mlp-collect PRIVATE kaivuri_mlp)
target_compile_definitions(kaivuri-mlp-collect PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" NATIVE_GIT_REVISION="${NATIVE_GIT_REVISION}")
add_library(kaivuri_collection src/ik_collection.cpp)
target_link_libraries(kaivuri_collection PUBLIC kaivuri_core)
add_library(kaivuri_pid_record src/pid_record.cpp)
target_link_libraries(kaivuri_pid_record PUBLIC kaivuri_collection kaivuri_mlp)
add_executable(kaivuri-pid-collect src/pid_main.cpp src/pid_options.cpp)
target_link_libraries(kaivuri-pid-collect PRIVATE kaivuri_pid_record)
target_compile_definitions(kaivuri-pid-collect PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" NATIVE_GIT_REVISION="${NATIVE_GIT_REVISION}")
add_executable(kaivuri-ik-preview src/ik_preview.cpp)
target_link_libraries(kaivuri-ik-preview PRIVATE kaivuri_collection kaivuri_mlp)
target_compile_definitions(kaivuri-ik-preview PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}" NATIVE_GIT_REVISION="${NATIVE_GIT_REVISION}")
add_executable(ik-collection-tests tests/ik_collection_tests.cpp)
target_link_libraries(ik-collection-tests PRIVATE kaivuri_collection)
add_executable(ik-reference-probe tests/ik_reference_probe.cpp)
target_link_libraries(ik-reference-probe PRIVATE kaivuri_collection)
if(BUILD_INFERENCE)
    find_package(CUDAToolkit REQUIRED)
    find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu REQUIRED)
    find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
    find_library(TENSORRT_PLUGIN_LIBRARY nvinfer_plugin REQUIRED)
    # The pi0.5 runtime, shared by kaivuri-inference and the ctypes library tools/vla_pi05.py loads.
    set_target_properties(kaivuri_core kaivuri_pi05 PROPERTIES POSITION_INDEPENDENT_CODE ON)
    add_library(kaivuri_pi05_trt STATIC src/pi05_runtime.cpp)
    set_target_properties(kaivuri_pi05_trt PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_include_directories(kaivuri_pi05_trt PUBLIC include ${TENSORRT_INCLUDE_DIR})
    target_link_libraries(kaivuri_pi05_trt PUBLIC kaivuri_pi05 yaml-cpp
        ${TENSORRT_LIBRARY} ${TENSORRT_PLUGIN_LIBRARY} CUDA::cudart)
    target_compile_options(kaivuri_pi05_trt PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    add_executable(kaivuri-inference src/inference_pi05.cpp)
    target_link_libraries(kaivuri-inference PRIVATE kaivuri_frames kaivuri_pi05_trt)
    target_compile_options(kaivuri-inference PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    add_library(kaivuri_pi05_runtime SHARED src/pi05_capi.cpp)
    target_link_libraries(kaivuri_pi05_runtime PRIVATE kaivuri_pi05_trt)
    target_compile_options(kaivuri_pi05_runtime PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()
if(BUILD_CAMERA)
    find_package(realsense2 2.58.4 EXACT CONFIG REQUIRED)
    add_executable(kaivuri-camera src/camera_main.cpp)
    target_link_libraries(kaivuri-camera PRIVATE kaivuri_core kaivuri_frames realsense2::realsense2 PkgConfig::ZSTD)
    target_compile_options(kaivuri-camera PRIVATE -Wall -Wextra -Wpedantic)
endif()

target_link_libraries(mlp-tests PRIVATE kaivuri_mlp)

target_compile_options(pi05-tests PRIVATE -ffp-contract=off)
target_compile_definitions(pi05-tests PRIVATE NATIVE_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME pi05-policy COMMAND pi05-tests $<TARGET_FILE:kaivuri-control>)

add_library(pwm-test-device SHARED tests/pwm_test_device.cpp)
target_link_libraries(pwm-test-device PRIVATE ${CMAKE_DL_LIBS})

if(ENABLE_SANITIZERS)
    execute_process(COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libasan.so OUTPUT_VARIABLE ASAN_RUNTIME OUTPUT_STRIP_TRAILING_WHITESPACE)
